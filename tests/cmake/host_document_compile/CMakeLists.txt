cmake_minimum_required(VERSION 3.29)
project(HostDocumentBoundaryProbe LANGUAGES CXX)
if(NOT MSVC OR NOT DEFINED LCNC_SOURCE_ROOT)
    message(FATAL_ERROR "Host 文档编译探针要求当前支持的 MSVC 工具链和源码根")
endif()
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
set(CMAKE_TRY_COMPILE_CONFIGURATION Debug)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /utf-8 /Zc:__cplusplus")
set(prefix "#include <lasercnc/kernel/app_kernel.hpp>\nusing namespace lasercnc;\n")
set(healthy "void probe(kernel::AppKernel& host, kernel::ProjectId p, kernel::DocumentId d) { (void)host.documentRuntime().create(p,d); (void)host.documentRuntime().open(d); (void)host.documents().snapshot(d); }\n")
set(attach "void probe(kernel::AppKernel& host, state::DocumentImage image) { (void)host.documentRuntime().attach(image); }\n")
set(attachRecovered "void probe(kernel::AppKernel& host, state::DocumentImage image) { (void)host.documentRuntime().attachRecovered(image); }\n")
set(adoptRecovered "void probe(kernel::AppKernel& host, std::vector<state::DocumentImage> images) { (void)host.documentRuntime().adoptRecovered(images); }\n")
set(restoreDocuments "void probe(state::DocumentStore& store, std::vector<state::DocumentImage> images) { (void)store.restoreDocuments(images); }\n")
foreach(mode IN ITEMS healthy attach attachRecovered adoptRecovered restoreDocuments)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${mode}.cpp" "${prefix}${${mode}}")
    try_compile(compiled SOURCES "${CMAKE_CURRENT_BINARY_DIR}/${mode}.cpp"
        CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${LCNC_SOURCE_ROOT}/include"
        CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON NO_CACHE OUTPUT_VARIABLE output)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${mode}.log" "compiled=${compiled}\n${output}")
    if(mode STREQUAL "healthy")
        if(NOT compiled)
            message(FATAL_ERROR "正常 Host 生命周期入口未编译成功；检查 healthy.log")
        endif()
    elseif(compiled)
        message(FATAL_ERROR "Host 不应能够编译镜像写入口 ${mode}；检查 ${mode}.log")
    else()
        string(REGEX MATCHALL "error C[0-9]+:[^\r\n]*" diagnostics "${output}")
        if(NOT diagnostics)
            message(FATAL_ERROR "${mode} 缺少实际 MSVC 编译错误；检查日志")
        endif()
        foreach(diagnostic IN LISTS diagnostics)
            if(NOT diagnostic MATCHES "error C(2039|2248):.*${mode}")
                message(FATAL_ERROR "${mode} 出现非预期编译错误：${diagnostic}")
            endif()
        endforeach()
    endif()
    message(STATUS "host-document-compile-verified: ${mode}")
endforeach()
